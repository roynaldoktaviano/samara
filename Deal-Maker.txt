/* v7.2: may the Meta browser cookies (_fbc/_fbp) be read/stored? Not for an EU/UK/CH (or region-unknown) denial. */
function gb_fb_cookie_ok($consent, $region) {
  if (strtolower(trim((string) $consent)) !== 'denied') return true;
  return !in_array(strtolower(trim((string) $region)), array('eu', 'uk', 'ch', ''), true);
}
function gb_fs_direct_log($row) {
  $log = get_option('gb_fs_direct_log', array());
  if (!is_array($log)) $log = array();
  $row['t'] = wp_date('Y-m-d H:i:s');
  $log[] = $row;
  if (count($log) > 300) $log = array_slice($log, -300);
  update_option('gb_fs_direct_log', $log, false);
}
function gb_fs_direct_enabled_for($form_id, $post_id) {
  $list = array_map('trim', explode(',', GB_FS_DIRECT_FORMS));
  return in_array('*', $list, true) || in_array($form_id . ':*', $list, true) || in_array($form_id . ':' . $post_id, $list, true);
}
function gb_dm_mode($src) {
  $c = array('form' => GB_DM_MODE_FORM, 'webhook' => GB_DM_MODE_WEBHOOK, 'wa' => GB_DM_MODE_WA);
  $m = get_option('gb_dm_mode_' . $src, '');
  if (!in_array($m, array('off', 'log', 'test', 'on'), true)) $m = isset($c[$src]) ? $c[$src] : 'off';
  return $m;
}
function gb_dm_is_test($email, $first) {
  return (bool) preg_match('/^abots\+[^@]+@samarayachting\.com$/i', (string) $email) && stripos(trim((string) $first), 'TEST') === 0;
}
function gb_dm_live($src, $is_test) { $m = gb_dm_mode($src); return $m === 'on' || ($m === 'test' && $is_test); }
function gb_dm_e164($p) {
  $p = trim((string) $p); if ($p === '') return '';
  $d = preg_replace('/[^\d+]/', '', $p);
  if (strpos($d, '00') === 0) $d = '+' . substr($d, 2);
  if (preg_match('/^\+[1-9]\d{6,14}$/', $d)) return $d;
  return '';
}

add_action('wpcf7_mail_sent', function ($cf) {
  try {
    if (!class_exists('WPCF7_Submission')) return;
    $sub = WPCF7_Submission::get_instance();
    if (!$sub || $sub->get_status() !== 'mail_sent') return;
    $form_id = (int) $cf->id();
    $post_id = (int) $sub->get_meta('container_post_id');
    if (!gb_fs_direct_enabled_for($form_id, $post_id)) return;
    if (!get_option('gb_fs_direct_key')) { gb_fs_direct_log(array('form' => $form_id, 'post' => $post_id, 'err' => 'no_key')); return; }
    gb_fs_form_queue($form_id, $post_id, (array) $sub->get_posted_data(), false);
  } catch (\Throwable $e) {
    gb_fs_direct_log(array('err' => 'prep_exception', 'msg' => substr($e->getMessage(), 0, 200)));
  }
}, 20);

    $contact = array('emails' => $email);
    foreach ($core as $src => $dst) { $v = sanitize_text_field($g($src)); if ($v !== '') $contact[$dst] = mb_substr($v, 0, 255); }
    $custom = array();
    $fb_ok = gb_fb_cookie_ok($g('ad_consent'), $g('consent_region'));   // v7.2: no _fbc/_fbp for an EU denial

      if ($dst === 'cf_details_and_special_requests') $v = sanitize_textarea_field($v); else $v = sanitize_text_field($v);
      if ($dst === 'cf_ad_consent' && !in_array($v, array('granted','denied','unknown','not_required'), true)) continue;

function gb_fs_direct_flush($sync = false) {
  if (empty($GLOBALS['gb_fs_direct_queue'])) return;
  try {
    if ($sync !== true) {
      if (function_exists('fastcgi_finish_request')) { @fastcgi_finish_request(); }
      elseif (function_exists('litespeed_finish_request')) { @litespeed_finish_request(); }
    }
    @ignore_user_abort(true);
    foreach ($GLOBALS['gb_fs_direct_queue'] as $job) {
      $t0 = microtime(true);
      $row = array('form' => $job['form'], 'post' => $job['post'], 'ip_set' => $job['ip'], 'consent' => $job['consent']);
      $email = $job['contact']['emails'];
      $live = gb_dm_live('form', !empty($job['enq']['is_test']));
      $existed = null;
      if (gb_dm_mode('form') !== 'off') {
        // Is this email already a contact? A NEW contact gets Form Source only when the deal maker is live for it (never overwrites).
        list($lc, $lj) = gb_fs_api('GET', 'lookup?f=email&entities=contact&q=' . rawurlencode($email), null, 5);
        $found = ($lc === 200) ? (array) ($lj['contacts']['contacts'] ?? array()) : array();
        $row['lookup'] = $lc; $existed = ($lc === 200) ? (count($found) > 0) : null;
        if ($live && $existed === false && GB_FS_FORM_SOURCE !== '') $job['contact']['custom_field']['cf_form_source'] = GB_FS_FORM_SOURCE;
      }
      list($code, $b) = gb_fs_api('POST', 'contacts/upsert', array('unique_identifier' => array('emails' => $email), 'contact' => $job['contact']), 5);
      $row['upsert'] = $code;
      $cid = isset($b['contact']['id']) ? $b['contact']['id'] : null; $row['cid'] = $cid;
      if ($code >= 300 || !$cid) { $row['err'] = 'upsert_http'; gb_fs_direct_log($row); continue; }
      list($nc) = gb_fs_api('POST', 'notes', array('note' => array('description' => $job['note'], 'targetable_type' => 'Contact', 'targetable_id' => $cid)), 5);
      $row['note'] = $nc;
      $row['ms'] = (int) round((microtime(true) - $t0) * 1000);
      gb_fs_direct_log($row);
      if (gb_dm_mode('form') !== 'off') {
        $e = $job['enq']; $e['cid'] = $cid;
        $e['new'] = ($code === 201) || ($existed === false);
        $e['by_us'] = ($code === 201);
        gb_dm_run($e);  // immediately, same run
      }
    }
  } catch (\Throwable $e) {
    gb_fs_direct_log(array('err' => 'flush_exception', 'msg' => substr($e->getMessage(), 0, 200)));
  }
  $GLOBALS['gb_fs_direct_queue'] = array();
}

/* ---------- v6: the central deal maker ---------- */
function gb_fs_api($method, $path, $body = null, $timeout = 8) {
  $h = array('Authorization' => 'Token token=' . get_option('gb_fs_direct_key'), 'Content-Type' => 'application/json', 'Accept' => 'application/json');
  $args = array('method' => $method, 'timeout' => $timeout, 'headers' => $h);
  if ($body !== null) $args['body'] = wp_json_encode($body);
  $r = wp_remote_request('https://samarayachting.myfreshworks.com/crm/sales/api/' . $path, $args);
  if (is_wp_error($r)) return array(0, null);
  return array((int) wp_remote_retrieve_response_code($r), json_decode(wp_remote_retrieve_body($r), true));
}
function gb_fs_type_of_text($s) {
  $s = strtolower((string) $s);
  if ($s === '') return '?';
  if (preg_match('/\bot\b|open trip|shared|cabin|kabine|last minute|#ot/', $s)) return 'OT';
  if (preg_match('/\bpc\b|private|mischief|otium|charter|family|honeymoon|diving|group/', $s)) return 'PC';
  return '?';
}
function gb_fs_product_interest($trip) {
  $s = strtolower((string) $trip); $t = gb_fs_type_of_text($s);
  if ($t === 'OT') return preg_match('/samara ii|\bs2\b/', $s) ? '' : 'OT / S1';
  if ($t === 'PC') {
    if (strpos($s, 'mischief') !== false) return 'PC / Mischief';
    if (strpos($s, 'otium') !== false) return 'PC / Otium';
    if (preg_match('/samara ii|\bs2\b/', $s)) return 'PC / S2';
    if (preg_match('/samara i\b|\bs1\b/', $s)) return 'PC / S1';
  }
  return '';
}
/* Round-robin pool, used ONLY when the contact has no owner. Inactive pool members are skipped (cached 6 h). */
function gb_fs_pool_active() {
  $ids = get_transient('gb_fs_active_users');
  if (!is_array($ids)) {
    list($c, $j) = gb_fs_api('GET', 'selector/owners');
    $ids = array();
    if ($c === 200 && is_array($j)) { foreach ((array) ($j['users'] ?? array()) as $u) if (!empty($u['is_active'])) $ids[] = (int) $u['id']; set_transient('gb_fs_active_users', $ids, 6 * HOUR_IN_SECONDS); }
  }
  $pool = array_values(array_filter(array_map('intval', explode(',', GB_FS_RR_POOL))));
  $act = $ids ? array_values(array_filter($pool, function ($u) use ($ids) { return in_array($u, $ids, true); })) : array();
  return $act ? $act : $pool;
}

function gb_fs_deal_fields($e, $existing_cf = null) {

  $nights = ($g('in') && $g('out') && strtotime($g('out')) > strtotime($g('in'))) ? (int) round((strtotime($g('out')) - strtotime($g('in'))) / 86400) : '';

    'cf_number_of_guestss' => ((int) $g('guests')) > 0 ? (int) $g('guests') : '',

    // Signature Voyage (#SL) -> 'SV / Otium', otherwise (Private Charter (#SL)) -> 'PC / Otium'. Brand stays Samara.
    'cf_product_interest' => in_array((int) $g('post'), array(53857, 53858, 55963, 55964, 55967, 53859), true)
      ? (stripos((string) $g('trip'), 'signature voyage') !== false ? 'SV / Otium' : 'PC / Otium')
      : (gb_fs_product_interest($g('trip')) !== '' ? gb_fs_product_interest($g('trip')) : $g('pi_page')),
    'cf_route' => $g('destination'), 'cf_budget' => ((int) $g('budget_usd')) > 0 ? (int) $g('budget_usd') : '', 'cf_private_charter_budget' => $g('budget'),
    'cf_gcl_id' => $g('gclid'), 'cf_gbraid' => $g('gbraid'), 'cf_wbraid' => $g('wbraid'), 'cf_fbclid' => $g('fbclid'), 'cf_msclkid' => $g('msclkid'),

function gb_fs_deal_retry($e, $row, $why) {
  $wait = array(120, 600, 3600);
  $row['err'] = $why; $row['attempt'] = $e['attempt'];
  if ($e['attempt'] < count($wait)) {
    $e['attempt']++;
    wp_schedule_single_event(time() + $wait[$e['attempt'] - 1], 'gb_fs_deal_retry_event', array($e));
    $row['retry_in_s'] = $wait[$e['attempt'] - 1];
  } else {
    $row['final'] = 'deal_failed';
    if (!empty($row['live'])) gb_fs_api('POST', 'notes', array('note' => array('targetable_type' => 'Contact', 'targetable_id' => $e['cid'],
      'description' => gb_fs_enquiry_note($e, 'DEAL COULD NOT BE CREATED AUTOMATICALLY - please create it by hand.'))));
  }
  gb_fs_direct_log($row);
  return $row;
}
add_action('gb_fs_deal_retry_event', function ($e) { gb_dm_run($e); }, 10, 1);

/* Per-contact lock (atomic add_option). Stale after 120 s. */
function gb_dm_lock($cid) {
  $k = 'gb_dm_lk_' . (int) $cid;
  if (add_option($k, time(), '', 'no')) return true;
  if ((int) get_option($k, 0) < time() - 120) { delete_option($k); return add_option($k, time(), '', 'no'); }
  return false;
}
function gb_dm_unlock($cid) { delete_option('gb_dm_lk_' . (int) $cid); }

/* A Sales-pipeline, not-closed deal of this contact created within the fresh window of the enquiry (e.g. the old workflow's deal). */
function gb_dm_find_fresh($e) {
  list($c, $j) = gb_fs_api('GET', 'contacts/' . (int) $e['cid'] . '?include=deals');
  if ($c !== 200 || !is_array($j)) return null;
  $stages = array(); foreach ((array) ($j['deal_stages'] ?? array()) as $s) $stages[$s['id']] = (string) ($s['forecast_type'] ?? '');
  $want = gb_fs_type_of_text($e['trip'] ?? '');
  foreach ((array) ($j['deals'] ?? array()) as $d) {
    if ((int) ($d['deal_pipeline_id'] ?? 0) !== GB_FS_PIPELINE) continue;
    if (stripos($stages[$d['deal_stage_id']] ?? '', 'closed') === 0) continue;
    $age = $e['ts'] - strtotime($d['created_at']);
    $dt = gb_fs_type_of_text(($d['custom_field']['cf_product_interest'] ?? '') . ' ' . ($d['name'] ?? ''));
    if ($age > -GB_FS_FRESH_SECONDS && $age < GB_FS_FRESH_SECONDS && ($dt === $want || $dt === '?' || $want === '?')) return $d;
  }
  return null;
}

/* The one rule, for every source. $e: src, ts, cid, new, is_test, attempt + enquiry values. Returns the log row. */
function gb_dm_run($e) {
  $live = gb_dm_live($e['src'], !empty($e['is_test']));
  $row = array('step' => 'deal', 'src' => $e['src'], 'form' => $e['form'] ?? null, 'cid' => $e['cid'], 'new' => !empty($e['new']), 'mode' => gb_dm_mode($e['src']), 'live' => $live, 'test' => !empty($e['is_test']), 'attempt' => $e['attempt']);
  if (gb_dm_mode($e['src']) === 'off') return $row;
  if (!gb_dm_lock($e['cid'])) {
    if ($e['attempt'] < 3) { $e['attempt']++; wp_schedule_single_event(time() + 60, 'gb_fs_deal_retry_event', array($e)); }
    $row['case'] = 'locked_retry_60s'; gb_fs_direct_log($row); return $row;
  }
  try {
    if (!empty($e['new']) && $e['attempt'] === 0 && empty($e['no_wait'])) sleep(3); // let assignment rules set the contact owner
    list($c, $j) = gb_fs_api('GET', 'contacts/' . (int) $e['cid'] . '?include=deals,owner');
    if ($c !== 200 || !is_array($j)) { gb_dm_unlock($e['cid']); return gb_fs_deal_retry($e, $row, 'get_contact_' . $c); }
    $deals = (array) ($j['deals'] ?? array()); $stages = array();
    foreach ((array) ($j['deal_stages'] ?? array()) as $s) $stages[$s['id']] = (string) ($s['forecast_type'] ?? '');
    $owner = (int) ($j['contact']['owner_id'] ?? 0);
    $ignore = array_map('intval', explode(',', GB_FS_PIPES_IGNORE));
    $want_type = gb_fs_type_of_text($e['trip'] ?? '');
    $fresh = null; $open_same = array(); $noted = null;
    $lock = get_transient('gb_fs_rd_' . (int) $e['cid']);
    foreach ($deals as $d) {
      $pipe = (int) ($d['deal_pipeline_id'] ?? 0);
      if (in_array($pipe, $ignore, true)) continue;
      if (stripos($stages[$d['deal_stage_id']] ?? '', 'closed') === 0) continue;   // Won/Lost: read only, never written
      $age = $e['ts'] - strtotime($d['created_at']);
      $dt = gb_fs_type_of_text(($d['custom_field']['cf_product_interest'] ?? '') . ' ' . ($d['name'] ?? ''));
      $same = ($dt === $want_type || $dt === '?' || $want_type === '?');
      if (!empty($e['fill_deal']) && (string) $d['id'] === (string) $e['fill_deal'] && $pipe === GB_FS_PIPELINE) { $fresh = $d; continue; }   // v7 backfill: fill this open deal
      if ((string) $lock === 'n' . $d['id'] && $same) { $noted = $d; continue; }   // already noted for this submit
      if ($same && $pipe === GB_FS_PIPELINE && ((string) $d['id'] === (string) $lock || ($age > -GB_FS_FRESH_SECONDS && $age < GB_FS_FRESH_SECONDS))) { if (!$fresh) $fresh = $d; continue; }
      if ($same) $open_same[] = $d;
    }
    $row['deals_seen'] = count($deals);
    if (!empty($e['fill_deal']) && (!$fresh || (string) $fresh['id'] !== (string) $e['fill_deal'])) { $row['case'] = 'fill_deal_not_open_or_not_found'; $row['deal'] = $e['fill_deal']; gb_dm_unlock($e['cid']); gb_fs_direct_log($row); return $row; }
    // Old workflow "Create deal on new contact" still fires on contact CREATE (Trip Type/Lead Source not empty), typically 4-90 s later.
    // If this contact was created recently by someone else (browser capture), or by us while the workflow is not yet narrowed,
    // wait for that deal and FILL it (never a 2nd deal). Polls ~65 s in-process, then one WP-Cron recheck at +180 s; only then create.
    if (!$fresh && $e['src'] === 'form') {
      $created = strtotime((string) ($j['contact']['created_at'] ?? '')) ?: 0;
      $recent = $created > 0 && ($e['ts'] - $created) < GB_FS_FRESH_SECONDS;
      $narrowed = (get_option('gb_dm_wf_narrowed', '0') === '1');
      if ($recent && (empty($e['by_us']) || !$narrowed)) {
        $row['wf_expected'] = true;
        if ($live) {
          if (empty($e['wf_polled'])) {
            foreach (array(8, 12, 20, 25) as $sl) {
              sleep($sl);
              $fresh = gb_dm_find_fresh($e);
              if ($fresh) { $row['wf_deal_after_s'] = time() - $e['ts']; break; }
            }
          }
          if (!$fresh && empty($e['wf_deferred'])) {
            $e['wf_polled'] = 1; $e['wf_deferred'] = 1;
            wp_schedule_single_event(time() + 180, 'gb_fs_deal_retry_event', array($e));
            $row['case'] = 'waiting_for_old_workflow_deal_recheck_180s'; gb_dm_unlock($e['cid']); gb_fs_direct_log($row); return $row;
          }
          if (!$fresh) { $row['wf_no_deal_after_wait'] = true; $e['new'] = true; }   // contact made minutes ago by a capture: name it as a new enquiry   // deferred recheck, still none -> the deal maker creates it
        }
      }
    }
    if ($fresh) {
      $row['case'] = 'same_submit_fill_empty'; $row['deal'] = $fresh['id'];
      list($c2, $jd) = gb_fs_api('GET', 'deals/' . $fresh['id']);
      if ($c2 !== 200) { gb_dm_unlock($e['cid']); return gb_fs_deal_retry($e, $row, 'get_deal_' . $c2); }
      $patch = gb_fs_deal_fields($e, (array) ($jd['deal']['custom_field'] ?? array()));
      $row['fields'] = array_keys($patch);
      if ($live && $patch) { list($c3) = gb_fs_api('PUT', 'deals/' . $fresh['id'], array('deal' => array('custom_field' => $patch))); $row['put'] = $c3; }
      if ($live && $e['src'] === 'wa' && !empty($e['tapdata'])) {
        list($cn) = gb_fs_api('POST', 'notes', array('note' => array('targetable_type' => 'Deal', 'targetable_id' => $fresh['id'],
          'description' => gb_fs_enquiry_note($e, 'WhatsApp chat linked to this deal (website tap data added; existing values were not changed).'))));
        $row['note'] = $cn;
      }
      if ($live) set_transient('gb_fs_rd_' . (int) $e['cid'], $fresh['id'], 1800);
      gb_dm_unlock($e['cid']); gb_fs_direct_log($row); return $row;
    }
    if ($noted && !$fresh) { $row['case'] = 'same_submit_already_noted'; $row['deal'] = $noted['id']; gb_dm_unlock($e['cid']); gb_fs_direct_log($row); return $row; }
    if ($open_same) {
      usort($open_same, function ($a, $b) { return strcmp((string) $b['updated_at'], (string) $a['updated_at']); });
      $d = $open_same[0]; $row['case'] = 'open_same_type_note'; $row['deal'] = $d['id'];
      if ($live) {
        list($cn) = gb_fs_api('POST', 'notes', array('note' => array('targetable_type' => 'Deal', 'targetable_id' => $d['id'],
          'description' => gb_fs_enquiry_note($e, 'RETURNING GUEST: new enquiry on this open deal. Values below are the NEW request; the deal was not changed.'))));
        list($ct) = gb_fs_api('POST', 'tasks', array('task' => array('title' => 'Returning guest sent a new enquiry: check dates/guests',
          'description' => gb_fs_enquiry_note($e, 'New enquiry from an existing guest (see deal note).'),
          'due_date' => gmdate('c', time() + 4 * 3600), 'owner_id' => $d['owner_id'], 'targetable_type' => 'Deal', 'targetable_id' => $d['id'])));
        $row['note'] = $cn; $row['task'] = $ct;
        if ($cn >= 300 || !$cn) { gb_dm_unlock($e['cid']); return gb_fs_deal_retry($e, $row, 'deal_note_' . $cn); }
        set_transient('gb_fs_rd_' . (int) $e['cid'], 'n' . $d['id'], 1800);
      }
      gb_dm_unlock($e['cid']); gb_fs_direct_log($row); return $row;
    }
    // New deal (new guest, or returning guest with no open deal of this type). Owner = contact owner; round-robin only if none.
    if ($owner > 0) { $row['owner'] = 'contact_owner'; }
    else {
      $owner = gb_fs_rr_owner($live); $row['owner'] = 'round_robin';
      if ($live) { list($co) = gb_fs_api('PUT', 'contacts/' . (int) $e['cid'], array('contact' => array('owner_id' => $owner))); $row['contact_owner_put'] = $co; }
    }
    $row['owner_id'] = $owner;
    $name = trim(($e['first'] ?? '') . ' ' . ($e['last'] ?? ''));
    if ($name === '') $name = trim(($j['contact']['first_name'] ?? '') . ' ' . ($j['contact']['last_name'] ?? ''));
    if ($name === '') $name = !empty($e['phone']) ? $e['phone'] : 'Contact ' . (int) $e['cid'];
    $deal = array('name' => $name . ' Deal' . (!empty($e['new']) ? '' : ' (re-enquiry ' . wp_date('j M Y', $e['ts']) . ')'), 'amount' => 0,
      'deal_pipeline_id' => GB_FS_PIPELINE, 'deal_stage_id' => GB_FS_STAGE_NEW, 'owner_id' => $owner,
      'contacts_added_list' => array((int) $e['cid']), 'custom_field' => gb_fs_deal_fields($e));
    $ls = (int) ($j['contact']['lead_source_id'] ?? 0); if ($ls > 0) $deal['lead_source_id'] = $ls;   // "Source", as the old workflow copied
    $row['case'] = !empty($e['new']) ? 'new_contact_new_deal' : 'returning_new_deal'; $row['fields'] = array_keys($deal['custom_field']);
    if (!$live) { gb_dm_unlock($e['cid']); gb_fs_direct_log($row); return $row; }
    list($cd, $jd) = gb_fs_api('POST', 'deals', array('deal' => $deal));
    if ($cd === 400 && stripos(wp_json_encode($jd), 'inactive') !== false) {
      // Freshsales refuses deals for a deactivated owner: fall back to round-robin (contact gets the same person), logged.
      $owner = gb_fs_rr_owner(true); $deal['owner_id'] = $owner; $row['owner'] = 'inactive_owner_fallback_round_robin'; $row['owner_id'] = $owner;
      list($co) = gb_fs_api('PUT', 'contacts/' . (int) $e['cid'], array('contact' => array('owner_id' => $owner))); $row['contact_owner_put'] = $co;
      list($cd, $jd) = gb_fs_api('POST', 'deals', array('deal' => $deal));
    }
    $row['deal_http'] = $cd; $row['deal'] = $jd['deal']['id'] ?? null;
    if (!$row['deal']) { gb_dm_unlock($e['cid']); return gb_fs_deal_retry($e, $row, 'deal_post_' . $cd); }
    set_transient('gb_fs_rd_' . (int) $e['cid'], $row['deal'], 1800);
    gb_fs_api('POST', 'notes', array('note' => array('targetable_type' => 'Deal', 'targetable_id' => $row['deal'],
      'description' => gb_fs_enquiry_note($e, !empty($e['new']) ? 'New enquiry (deal created by the central deal maker).' : 'New deal for a returning guest (earlier deals untouched).'))));
    if (($e['match'] ?? '') === 'Low') {
      list($ct) = gb_fs_api('POST', 'tasks', array('task' => array('title' => 'LOW confidence WhatsApp match: please check by hand',
        'description' => gb_fs_enquiry_note($e, 'This deal was matched only by timing, and 2 or more website taps fit, so it could be the wrong guest. Check the match. If it is right, correct the click IDs if needed, then change Click ID match to Medium or Confident so it can be uploaded to Google and Meta. While this is Low it will not be uploaded.'),
        'due_date' => gmdate('c', time() + 4 * 3600), 'owner_id' => $owner, 'targetable_type' => 'Deal', 'targetable_id' => $row['deal'])));
      $row['task'] = $ct;
    }
    gb_dm_unlock($e['cid']); gb_fs_direct_log($row); return $row;
  } catch (\Throwable $ex) {
    gb_dm_unlock($e['cid']);
    return gb_fs_deal_retry($e, $row, 'deal_step_exception: ' . substr($ex->getMessage(), 0, 150));
  }
}

/* Build an enquiry from a Freshsales contact (webhook / WhatsApp sources). */
function gb_dm_enq_from_contact($ct, $src) {
  $cf = (array) ($ct['custom_field'] ?? array());
  $v = function ($k, $len = 255) use ($cf) { $x = isset($cf[$k]) ? trim((string) $cf[$k]) : ''; return mb_substr($x, 0, $len); };
  $iso = function ($x) { return preg_match('/^\d{4}-\d{2}-\d{2}/', (string) $x) ? substr($x, 0, 10) : ''; };
  $email = (string) ($ct['email'] ?? '');
  $phone_raw = $v('cf_whatsapp_number') !== '' ? $v('cf_whatsapp_number') : (string) ($ct['mobile_number'] ?? '');
  $granted = gb_fb_cookie_ok($v('cf_ad_consent'), $v('cf_consent_region'));   // v7.2: copy contact _fbc to the deal unless EU denial
  $host = '';
  foreach (array('cf_e_mail_adresse', 'cf_telefon_wa_lead_field', 'cf_mobile_number_backup') as $k) { $h = wp_parse_url($v($k), PHP_URL_HOST); if ($h) { $host = preg_replace('/^www\./', '', strtolower($h)); break; } }
  $created = strtotime((string) ($ct['created_at'] ?? '')) ?: 0;
  $e = array('src' => $src, 'ts' => time(), 'cid' => (int) $ct['id'], 'attempt' => 0, 'email' => $email,
    'new' => $created > 0 && (time() - $created) < 1800, 'first' => (string) ($ct['first_name'] ?? ''), 'last' => (string) ($ct['last_name'] ?? ''),
    'trip' => $v('cf_trip_type'), 'in' => $iso($v('cf_trip_date')), 'out' => $iso($v('cf_check_out_date')),
    'guests' => (int) preg_replace('/\D/', '', $v('cf_number_of_guests')), 'request' => $v('cf_details_and_special_requests', 9000),
    'destination' => $v('cf_destination'), 'budget' => $v('cf_private_charter_budget'),
    'phone' => gb_dm_e164($phone_raw) !== '' ? gb_dm_e164($phone_raw) : mb_substr($phone_raw, 0, 255),
    'gclid' => $v('cf_gclid'), 'gbraid' => $v('cf_gbraid'), 'wbraid' => $v('cf_wbraid'), 'fbclid' => $v('cf_fbclid'), 'msclkid' => $v('cf_interest'),
    'utm_source' => $v('cf_utm_source'), 'utm_medium' => $v('cf_utm_medium'), 'utm_campaign' => $v('cf_utm_campaign'),
    'landing' => $v('cf_telefon_wa_lead_field'), 'website' => $host, 'channel' => $src === 'wa' ? 'WhatsApp' : 'Form',
    'fbc' => $granted ? $v('cf_wa_chat') : '', 'lead_source' => $v('cf_lead_source'), 'form_source' => $v('cf_form_source'),
  );
  $e['match'] = ($e['gclid'] . $e['gbraid'] . $e['wbraid'] . $e['fbclid'] . $e['msclkid'] . $e['fbc']) !== '' ? 'Confident' : '';
  $e['is_test'] = gb_dm_is_test($email, $e['first']);
  return $e;
}
/* Which brand/site does a contact belong to? Only Samara is in scope; others stay with the old workflow for now. */
function gb_dm_is_samara($e) {
  if ($e['website'] !== '') return $e['website'] === GB_DM_SITE;
  $t = strtolower($e['trip'] . ' ' . $e['form_source']);
  if (preg_match('/otium|mischief ?voyage|siloina|silolona/', $t) && strpos($t, '#sl') === false) return false;
  return strpos($t, '#sl') !== false || $e['src'] === 'wa' ? true : null;   // null = unknown
}
function gb_dm_wa_tap($ref) {
  global $wpdb; if (!function_exists('gb_wa_table')) return null;
  return $wpdb->get_row($wpdb->prepare('SELECT * FROM ' . gb_wa_table() . ' WHERE ref = %s LIMIT 1', $ref), ARRAY_A);
}
function gb_dm_wa_taps_before($ts, $test_only = false) {
  global $wpdb; if (!function_exists('gb_wa_table')) return array();
  // TEST records only see TEST taps (utm_source = dmtest), so tests never pick up a real visitor's tap; real records never see TEST taps.
  $w = $test_only ? " AND utm_source = 'dmtest'" : " AND (utm_source IS NULL OR utm_source <> 'dmtest')";
  return (array) $wpdb->get_results($wpdb->prepare('SELECT * FROM ' . gb_wa_table() . ' WHERE created_at BETWEEN %s AND %s' . $w . ' ORDER BY id LIMIT 10',
    gmdate('Y-m-d H:i:s', $ts - GB_DM_WA_WINDOW), gmdate('Y-m-d H:i:s', $ts)), ARRAY_A);
}

/* v7: every value stored on the tap -> enquiry. Deal fields where they exist; the rest in $e['tapdata'] (structured note). */
function gb_dm_pi_from_page($url) {
  $p = strtolower((string) wp_parse_url((string) $url, PHP_URL_PATH));
  if ($p === '') return '';
  if (preg_match('#/(shared-trip|open-trip|thankyou-ot)(/|$)#', $p)) return 'OT / S1';
  if (preg_match('#samara-ii#', $p)) return 'PC / S2';
  if (preg_match('#samara-i(-|/|$)#', $p)) return 'PC / S1';
  return '';
}
function gb_dm_apply_tap(&$e, $tap, $level) {
  foreach (array('gclid', 'gbraid', 'wbraid', 'fbclid', 'msclkid', 'utm_source', 'utm_medium', 'utm_campaign') as $k) $e[$k] = (string) ($tap[$k] ?? '');
  // v7.2 (Marc 15:07/15:10): stored for every tap whatever the consent, except Meta cookies on an EU denial; uploads apply consent.
  $fb_ok = gb_fb_cookie_ok($tap['ad_consent'] ?? '', $tap['consent_region'] ?? '');   // Meta cookies: not for an EU denial (Marc 15:10)
  $e['fbc'] = $fb_ok ? (string) ($tap['_fbc'] ?? '') : '';
  $e['fbp'] = $fb_ok ? (string) ($tap['_fbp'] ?? '') : '';
  $e['ua'] = mb_substr((string) ($tap['ua'] ?? ''), 0, 2000);
  $e['landing'] = mb_substr((string) ($tap['page'] ?? ''), 0, 255);
  if (($e['trip'] ?? '') === '') $e['pi_page'] = gb_dm_pi_from_page($tap['page'] ?? '');
  $t = strtotime((string) ($tap['created_at'] ?? '') . ' UTC') ?: 0;
  $e['tap_consent'] = (string) ($tap['ad_consent'] ?? ''); $e['tap_region'] = (string) ($tap['consent_region'] ?? '');
  $e['tapdata'] = array(
    'match_confidence' => $level, 'tap_ref' => (string) ($tap['ref'] ?? ''),
    'tap_time_utc' => (string) ($tap['created_at'] ?? ''), 'tap_time_wita' => $t ? wp_date('Y-m-d H:i:s', $t) . ' WITA' : '',
    'ad_consent' => (string) ($tap['ad_consent'] ?? ''), 'consent_region' => (string) ($tap['consent_region'] ?? ''),
    'consent_recorded_at' => $t ? wp_date('Y-m-d H:i:s', $t) . ' WITA (consent state read by the tap script at the tap)' : '',
    'page_url' => (string) ($tap['page'] ?? ''), 'language' => (string) ($tap['lang'] ?? ''), 'samara_visitor_id' => (string) ($tap['vid'] ?? ''),
    'gclid' => (string) ($tap['gclid'] ?? ''), 'gbraid' => (string) ($tap['gbraid'] ?? ''), 'wbraid' => (string) ($tap['wbraid'] ?? ''),
    'fbclid' => (string) ($tap['fbclid'] ?? ''), 'msclkid' => (string) ($tap['msclkid'] ?? ''), 'fbc' => $fb_ok ? (string) ($tap['_fbc'] ?? '') : '(not read: EU consent denied)', 'fbp' => $fb_ok ? (string) ($tap['_fbp'] ?? '') : '(not read: EU consent denied)',
    'ip' => (string) ($tap['ip'] ?? ''), 'referrer' => (string) ($tap['referrer'] ?? ''), 'user_agent' => (string) ($tap['ua'] ?? ''),
    'utm_source' => (string) ($tap['utm_source'] ?? ''), 'utm_medium' => (string) ($tap['utm_medium'] ?? ''), 'utm_campaign' => (string) ($tap['utm_campaign'] ?? ''),
    'utm_content' => (string) ($tap['utm_content'] ?? ''), 'utm_term' => (string) ($tap['utm_term'] ?? ''),
    'whatsapp_chat_started' => !empty($e['chat_at']) ? $e['chat_at'] : '', 'freshchat_conversation' => !empty($e['fc_conv']) ? $e['fc_conv'] : '',
  );
  if ((string) ($tap['ad_consent'] ?? '') === 'denied') $e['upload'] = 'Consent DENIED on the website tap (region ' . ((string) ($tap['consent_region'] ?? '') ?: 'unknown') . '): NEVER upload this guest to Google or Meta.';
}
/* v7: tap consent -> Freshsales contact. Fill empty; a denial always wins. Only for Confident/Medium (Low may be the wrong guest). */
function gb_dm_contact_consent($cid, $ct, $e, $live) {
  $v = (string) ($e['tap_consent'] ?? ''); $rg = (string) ($e['tap_region'] ?? '');
  if (!in_array($v, array('granted', 'denied', 'not_required'), true)) return array('consent' => 'tap_' . ($v ?: 'empty') . '_not_written');
  $cf = (array) ($ct['custom_field'] ?? array()); $cur = trim((string) ($cf['cf_ad_consent'] ?? '')); $curr = trim((string) ($cf['cf_consent_region'] ?? ''));
  $put = array();
  if ($cur === '' || ($v === 'denied' && $cur !== 'denied')) $put['cf_ad_consent'] = $v;
  if ($rg !== '' && ($curr === '' || ($v === 'denied' && $cur !== 'denied'))) $put['cf_consent_region'] = $rg;
  if (!$put) return array('consent' => 'contact_already_set', 'contact_value' => $cur);
  if (!$live) return array('consent' => 'would_write', 'fields' => array_keys($put));
  // Freshsales refuses any contact update without an email or External ID (400 "Fill any one field"); WhatsApp-only contacts have neither.
  if (trim((string) ($ct['email'] ?? '')) === '' && empty($ct['emails']) && trim((string) ($ct['external_id'] ?? '')) === '') return array('consent' => 'not_written_contact_has_no_email', 'fields' => array_keys($put), 'note' => 'consent is in the deal note');
  list($c) = gb_fs_api('PUT', 'contacts/' . (int) $cid, array('contact' => array('custom_field' => $put)));
  return array('consent' => 'written', 'http' => $c, 'fields' => array_keys($put), 'was' => $cur);
}

  $req->set_header('x-gb-secret', (string) get_option('gb_dm_secret', ''));
  $req->set_header('content-type', 'application/json');
  $body = array('source' => 'wa', 'phone' => $phone, 'message' => (string) $text, 'chat_started_at' => $msg_time, 'fc_conv' => (string) $conv);
  if ($ref !== '') $body['ref'] = $ref;
  if ($test_only) $body['test_only'] = 1;
  $req->set_body(wp_json_encode($body));
  $res = rest_do_request($req); $d = (array) $res->get_data();
  $row['case'] = $ref !== '' ? 'sl_code_sent_to_deal_maker' : 'no_code_sent_to_deal_maker';
  $row['http'] = $res->get_status(); $row['dm'] = $d['case'] ?? ($d['err'] ?? null);
  $row['cid'] = $d['cid'] ?? null; $row['live'] = $d['live'] ?? null; $row['match'] = $d['match'] ?? null;
  return array($row, !in_array(($d['err'] ?? ''), array('no_contact', 'exception'), true));
}
function gb_dm_fc_poll() {
  if (get_option('gb_dm_fc_poll', '1') !== '1' || gb_dm_mode('wa') === 'off') return array('skipped' => 'off');
  if (strlen((string) get_option('gb_dm_fc_key', '')) < 20) return array('skipped' => 'no_key');
  if (get_transient('gb_dm_fc_running')) return array('skipped' => 'running');
  set_transient('gb_dm_fc_running', 1, 240);
  $now = time(); $st = (array) get_option('gb_dm_fc_status', array());
  $since = max($now - 7200, (int) ($st['last_ok'] ?? 0) - 900);              // overlap 15 min, never older than 2 h
  $seen = (array) get_option('gb_dm_fc_seen', array());
  $out = array('users' => 0, 'convs' => 0, 'decided' => 0, 'err' => null);
  try {
    for ($pg = 1; $pg <= 3; $pg++) {
      list($c, $j) = gb_dm_fc_get('users', array('created_from' => '2015-01-01T00:00:00Z', 'created_to' => gmdate('Y-m-d\TH:i:s\Z', $now + 60),
        'updated_from' => gmdate('Y-m-d\TH:i:s\Z', $since), 'items_per_page' => 100, 'page' => $pg));
      if ($c !== 200) { $out['err'] = 'users_' . $c; break; }
      foreach ((array) ($j['users'] ?? array()) as $u) {
        $out['users']++;
        $phone = gb_dm_e164($u['phone'] ?? '');
        list($cc, $cj) = gb_dm_fc_get('users/' . rawurlencode($u['id']) . '/conversations');
        if ($cc !== 200) continue;
        foreach ((array) ($cj['conversations'] ?? array()) as $cv) {
          $id = (string) ($cv['id'] ?? ''); if ($id === '') continue;
          if (isset($seen[$id]['ch']) && $seen[$id]['ch'] !== GB_DM_FC_CHANNEL) continue;   // known non-WhatsApp conversation
          if (!isset($seen[$id]['ch'])) {
            list($dc, $dj) = gb_dm_fc_get('conversations/' . rawurlencode($id)); if ($dc !== 200) continue;
            $seen[$id] = array('ch' => (string) ($dj['channel_id'] ?? ''), 'm' => '', 'n' => 0, 't' => $now);
            if ($seen[$id]['ch'] !== GB_DM_FC_CHANNEL) continue;
          }
          $out['convs']++;
          list($mc, $mj) = gb_dm_fc_get('conversations/' . rawurlencode($id) . '/messages', array('items_per_page' => 50));
          if ($mc !== 200) continue;
          $um = array_values(array_filter((array) ($mj['messages'] ?? array()), function ($m) { return ($m['actor_type'] ?? '') === 'user'; }));
          usort($um, function ($a, $b) { return strcmp((string) $a['created_time'], (string) $b['created_time']); });
          $prev = 0;
          foreach ($um as $m) {
            $mt = (string) $m['created_time']; $ts = strtotime($mt) ?: 0;
            $session_start = ($prev === 0 || $ts - $prev >= 6 * 3600); $prev = $ts;
            if ($mt <= (string) $seen[$id]['m'] || $ts < $now - 7200) continue;   // already handled, or older than 2 h
            if ($session_start) {
              list($row, $done) = gb_dm_fc_decide($id, $mt, gb_dm_fc_text($m), $phone);
              if (!$done && (int) $seen[$id]['n'] < 6) { $seen[$id]['n']++; $row['retry'] = $seen[$id]['n']; gb_fs_direct_log($row); $out['decided']++; break; }
              gb_fs_direct_log($row); $out['decided']++; $seen[$id]['n'] = 0;
            }
            $seen[$id]['m'] = $mt; $seen[$id]['t'] = $now;
          }
        }
      }
      if ($pg >= (int) ($j['pagination']['total_pages'] ?? 1)) break;
    }
  } catch (\Throwable $ex) { $out['err'] = 'exception: ' . substr($ex->getMessage(), 0, 120); }
  foreach ($seen as $k => $v) if ((int) ($v['t'] ?? 0) < $now - 14 * 86400) unset($seen[$k]);
  update_option('gb_dm_fc_seen', $seen, false);
  $st = array('last_run' => wp_date('Y-m-d H:i:s', $now), 'last_ok' => $out['err'] ? (int) ($st['last_ok'] ?? 0) : $now, 'last' => $out);
  update_option('gb_dm_fc_status', $st, false);
  delete_transient('gb_dm_fc_running');
  return $out;
}
/* Admin-only: run the poll now, or simulate one SL- chat for a TEST contact (never messages anyone; TEST contacts only). */
add_action('rest_api_init', function () {
  register_rest_route('grokfix/v1', '/fc-poll', array('methods' => 'POST', 'permission_callback' => function () { return current_user_can('manage_options'); },
    'callback' => function ($r) {
      $p = (array) $r->get_json_params();
      if (!empty($p['sim'])) {
        $s = (array) $p['sim'];
        list($row, $done) = gb_dm_fc_decide('TEST-' . substr((string) ($s['conv'] ?? 'sim'), 0, 30), (string) ($s['at'] ?? gmdate('Y-m-d\TH:i:s\Z')), (string) ($s['text'] ?? ''), gb_dm_e164($s['phone'] ?? ''), true);
        $row['sim'] = true; gb_fs_direct_log($row);
        return array('ok' => true, 'row' => $row, 'done' => $done);
      }
      return array('ok' => true, 'poll' => gb_dm_fc_poll());
    }));
});
